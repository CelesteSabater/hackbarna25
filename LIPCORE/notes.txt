*** afegir ngrok al PATH: (C:\Users\Joan\AppData\Local\Microsoft\WindowsApps\ngrok.exe)

   $ngrokDir = "C:\Users\Joan\AppData\Local\Microsoft\WindowsApps"
   $oldPath = [Environment]::GetEnvironmentVariable("PATH","User")
   if ($oldPath -notlike "*$ngrokDir*") {
     [Environment]::SetEnvironmentVariable("PATH", "$oldPath;$ngrokDir", "User")
     Write-Host "Añadido $ngrokDir al PATH de usuario. Cierra y vuelve a abrir PowerShell para que surta efecto."
   } else {
     Write-Host "$ngrokDir ya estaba en PATH de usuario."
   }
